    //创建ab对比文件

    public void CreateABCompareFile()

    {

        //获取文件夹信息

        //根据选择的平台，选择对应平台下的内容，来进行对比文件的生成

        DirectoryInfo directory = Directory.CreateDirectory(Application.dataPath + "/ArtRes/AB/"+targetStrs[nowSelIndex]);

        FileInfo[] fileInfos = directory.GetFiles();

        //用于存储信息的字符串

        string abCompareInfo = "";

        //遍历文件夹下的所有文件信息

        foreach (FileInfo info in fileInfos)

        {

            //没有后缀的 才是AB包 我们只想要AB包的信息

            if(info.Extension == "")

            {

                Debug.Log("文件名：" + info.Name);

                Debug.Log("文件路径：" + info.FullName);

                abCompareInfo += info.Name + " " + info.Length + " " + GetMD5(info.FullName);

                abCompareInfo += "\n";

            }

        }

        //去除最后一个换行符

        abCompareInfo = abCompareInfo.Substring(0, abCompareInfo.Length - 1);

        //将生成的对比信息保存到文件中

        //存储拼接好的 AB包资源信息

        File.WriteAllText(Application.dataPath + "/ArtRes/AB/"+targetStrs[nowSelIndex]+"/ABCompareInfo.txt", abCompareInfo);

        Debug.Log($"AB包对比文件生成成功");

        //刷新编辑器

        AssetDatabase.Refresh();

    }

    //获取文件的MD5码

    public string GetMD5(string filePath)

    {

        using(FileStream file = new FileStream(filePath, FileMode.Open))

        {

            //声明一个MD5对象 用于生成MD5码

            MD5 md5 = new MD5CryptoServiceProvider();

            //利用md5对象根据流信息 计算出MD5码（16字节的字节数组形式）

            byte[] md5Bytes = md5.ComputeHash(file);

  

            //关闭文件流

            file.Close();

  

            //将字节数组形式的MD5码转为 16进制字符串，为了减少md5码的长度

            StringBuilder sb = new StringBuilder();

            for (int i = 0; i < md5Bytes.Length; i++)

            {

                sb.Append(md5Bytes[i].ToString("x2"));

            }

            return sb.ToString();

        }

    }

  

    //添加默认资源

    public void MoveABToStreamAsset()

    {

        // 通过编辑器Selection类，获取所有选中的资源

        UnityEngine.Object[] selectedAssets = Selection.GetFiltered(typeof(UnityEngine.Object),SelectionMode.DeepAssets);

        //如果一个资源都没选，结束

        if (selectedAssets.Length == 0)

            return;

        //用于拼接本地默认AB包资源信息的字符串

        string abCompareInfo = "";

        //遍历选中资源对象

        foreach (UnityEngine.Object asset in selectedAssets)

        {

            //获取资源路径

            string assetPath = AssetDatabase.GetAssetPath(asset);

            //获取资源名称

            string assetName = assetPath.Substring(assetPath.LastIndexOf("/") + 1);

            //如果资源名称包含点号，说明不是AB包

            //还可以通过在拷贝之前获取全路径，然后通过FileInfo去获取文件后缀名，这样判断会更安全

            if(assetName.IndexOf(".") != -1)

                continue;

            //拼接目标路径

            string targetPath = "Assets/StreamingAssets/" + assetName;

            //复制资源到StreamingAssets目录下

            //若不存在StreamingAssets目录，则会失败

            AssetDatabase.CopyAsset(assetPath, targetPath);

            //获取拷贝后的文件信息

            FileInfo fileInfo = new FileInfo(Application.streamingAssetsPath + "/" + assetName);

            //拼接AB包信息

            abCompareInfo += assetName + " " + fileInfo.Length + " " + CreateABCompare.GetMD5(fileInfo.FullName);

            //分割多个AB包信息

            abCompareInfo += "\n";

        }

        //去除最后一个换行符

        abCompareInfo = abCompareInfo.Substring(0, abCompareInfo.Length - 1);

        //将生成的对比信息保存到文件中

        File.WriteAllText(Application.streamingAssetsPath + "/ABCompareInfo.txt", abCompareInfo);

  

        Debug.Log("移动AB包到StreamingAssets目录完成！");

        //刷新资源数据库

        AssetDatabase.Refresh();

    }

  

    public void UploadAllABFile()

    {

        //获取文件夹信息

        DirectoryInfo directory = Directory.CreateDirectory(Application.dataPath + "/ArtRes/AB/"+targetStrs[nowSelIndex]);

        FileInfo[] fileInfos = directory.GetFiles();

        //遍历文件夹下的所有文件信息

        foreach (FileInfo info in fileInfos)

        {

            //获取AB包文件

            //和资源对比文件（.txt,若想要更安全，使用文件名称匹配）

            if(info.Extension == "" || info.Extension == ".txt")

            {

                //上传文件

                FtpUploadFile(info.FullName, info.Name);

            }

        }

    }

  

    private async void FtpUploadFile(string filePath,string fileName)

    {

        //异步上传文件，异步函数相关知识点在进阶知识的C#补充中

        await Task.Run(() =>

        {

            //上传文件的逻辑

            //1，创建ftp链接，用于上传

            FtpWebRequest request = (FtpWebRequest)FtpWebRequest.Create(new Uri("ftp://10.100.209.199/AB/"+targetStrs[nowSelIndex]+"/"+fileName));

  

            //2，设置一个通信凭证，这样才能上传

            NetworkCredential credential = new NetworkCredential("XC", "XC12345");

            request.Credentials = credential;

            //3，其他设置

            //设置代理为null

            request.Proxy = null;

            //请求完毕后，是否关闭控制链接

            request.KeepAlive = false;

            //操作命令-上传

            request.Method = WebRequestMethods.Ftp.UploadFile;

            //指定传输的类型 2进制

            request.UseBinary = true;

            //4，上传文件

            //ftp的流对象

            Stream uploadStream = request.GetRequestStream();

  

            try

            {

                //读取文件信息 写入该流对象

                using(FileStream fileStream = File.OpenRead(filePath))

                {

                    //定义一个缓冲区

                    byte[] buffer = new byte[2048];

                    //返回实际读取的字节数

                    int contentLength = fileStream.Read(buffer, 0, buffer.Length);

                    //循环读取文件流，直到文件结束

                    while(contentLength != 0)

                    {

                        uploadStream.Write(buffer, 0, contentLength);

                        contentLength = fileStream.Read(buffer, 0, buffer.Length);

                    }

                    //关闭流对象

                    fileStream.Close();

                    uploadStream.Close();

                }

                Debug.Log(fileName + "上传成功");

            }

            catch(Exception e)

            {

                Debug.LogWarning(fileName + "上传失败"+e.Message);

            }

        });

    }

}
