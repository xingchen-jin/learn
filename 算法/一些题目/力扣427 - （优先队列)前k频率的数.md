class Solution {

    struct Cmp {

        bool operator()(const std::pair<int, int>& a, const std::pair<int, int>& b) const {

            return a.second < b.second;   // 按频次最大堆

        }

    };

  

public:

    vector<int> topKFrequent(vector<int>& nums, int k) {

        unordered_map<int, int> hashtable; //数字 -> 次数

        int n = nums.size();

        priority_queue<pair<int,int> , vector<pair<int,int>> , Cmp> que;

  

        //先遍历，得到每个数的次数

        for(int i=0;i<n;i++){

            hashtable[nums[i]]++;

        }

  

        //遍历哈希表，存入优先队列

        for(auto it : hashtable){

            que.push({it.first, it.second});

        }

  

        vector<int> res(k);

        for(int i=0;i<k;i++){

            res[i] = que.top().first;

            que.pop();

        }

        return res;

    }

};
